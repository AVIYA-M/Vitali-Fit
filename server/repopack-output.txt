import Activity from '../models/Activity.js';
import User from '../models/User.js';
import { calculateCalories } from '../utils/helpers.js';
import { ACTIVITY_MET } from '../constants/index.js';

// הוספת פעילות עצמאית (סוג ומשך כבר נבדקו ב-Joi)
export const addActivity = async (req, res, next) => {
  try {
    const userId = req.user.userId;
    const { activityType, durationMinutes, date } = req.body;

    const user = await User.findById(userId);
    if (!user) {
      return res.status(404).json({ message: 'המשתמש לא נמצא.' });
    }

    const selectedDate = date ? new Date(date) : new Date();

    if (selectedDate.getDay() === 6) {
      return res.status(400).json({ message: 'לא ניתן להוסיף פעילות בשבת.' });
    }

    // אם אין למשתמש משקל, נשתמש ב-70 ק"ג כברירת מחדל כדי לא לקבל NaN
    const userWeight = user.weight ? Number(user.weight) : 70;

    const caloriesBurned = calculateCalories(
      userWeight,
      durationMinutes,
      ACTIVITY_MET[activityType]
    );

    const activity = await Activity.create({
      userId,
      activityType,
      durationMinutes,
      caloriesBurned,
      date: selectedDate
    });

    res.status(201).json({
      message: 'הפעילות נוספה בהצלחה.',
      activity
    });

  } catch (error) {
    next(error);
  }
};

      return res.status(400).json({ message: 'אימייל או סיסמה שגויים.' });

    res.status(200).json({
      message: 'התחברת בהצלחה!',
      token: signToken(user),
      user: publicUser(user)
    });

  } catch (error) {
    next(error);
  }
};

import Tracking from '../models/Tracking.js';
import Workout from '../models/Workout.js';
import Activity from '../models/Activity.js';

import User from '../models/User.js';
import WeightHistory from '../models/WeightHistory.js';

// ========================================
// סימון אימון כהושלם
// ========================================
export const completeWorkout = async (req, res, next) => {
    try {
        const userId = req.user.userId;
        const workoutId = req.params.workoutId;

        const workout = await Workout.findById(workoutId);

        if (!workout) {
            return res.status(404).json({
                message: 'האימון לא נמצא.'
            });
        }

        const isRegistered =
            Array.isArray(workout.registeredUsers) &&
            workout.registeredUsers.some(
                id => String(id) === String(userId)
            );

        if (!isRegistered) {
            return res.status(403).json({
                message: 'אפשר לסמן כהושלם רק אימון שנרשמת אליו.'
            });
        }

        const existingTracking = await Tracking.findOne({
            userId,
            workoutId
        });

        if (existingTracking) {
            return res.status(400).json({
                message: 'האימון כבר סומן כהושלם.'
            });
        }

        const tracking = new Tracking({
            userId,
            workoutId,
            completedAt: new Date(),
            durationMinutes: Number(workout.durationMinutes) || 60,
            caloriesBurned: Number(workout.estimatedCalories) || 250
        });

        const savedTracking = await tracking.save();

        res.status(201).json({
            message: 'האימון סומן כהושלם בהצלחה.',
            tracking: savedTracking
        });

    } catch (error) {
        next(error);
    }
};

export const getTrackingDashboard = async (req, res, next) => {
    try {
        const userId = req.user.userId;

        const user = await User.findById(userId);

        if (!user) {
            return res.status(404).json({
                message: 'המשתמש לא נמצא.'
            });
        }

        const now = new Date();

        // ========================================
        // שבוע
        // ========================================
        const startOfWeek = getStartOfWeek(now);

        const endOfWeek = new Date(startOfWeek);
        endOfWeek.setDate(endOfWeek.getDate() + 7);


        // ========================================
        // אימונים
        // ========================================
        const weeklyWorkouts = await Tracking.find({
            userId,
            completedAt: {
                $gte: startOfWeek,
                $lt: endOfWeek
            }
        });

        const workoutCalories = weeklyWorkouts.reduce(
            (total, record) => {
                return total + (Number(record.caloriesBurned) || 0);
            },
            0
        );

        const workoutMinutes = weeklyWorkouts.reduce(
            (total, record) => {
                return total + (Number(record.durationMinutes) || 0);
            },
            0
        );


        // ========================================
        // פעילות נוספת
        // ========================================
        const weeklyActivities = await Activity.find({
            userId,
            date: {
                $gte: startOfWeek,
                $lt: endOfWeek
            }
        }).sort({
            date: -1
        });

        const activityCalories = weeklyActivities.reduce(
            (total, activity) => {
                return total + (Number(activity.caloriesBurned) || 0);
            },
            0
        );

        const activityMinutes = weeklyActivities.reduce(
            (total, activity) => {
                return total + (Number(activity.durationMinutes) || 0);
            },
            0
        );


        // ========================================
        // אוכל
        // ========================================
        const weeklyMeals = await Nutrition.find({
            userId,
            date: {
                $gte: startOfWeek,
                $lt: endOfWeek
            }
        });

        const foodCalories = weeklyMeals.reduce(
            (total, meal) => {
                return total + (Number(meal.calories) || 0);
            },
            0
        );


        // ========================================
        // סיכום קלורי
        // ========================================
        const totalBurnedCalories =
            workoutCalories +
            activityCalories;

        const calorieBalance =
            totalBurnedCalories -
            foodCalories;


        // ========================================
        // משקל התחלתי והפרשים
        // ========================================
        const weightHistory = await WeightHistory.find({
            userId
        }).sort({
            date: 1,
            _id: 1
        });

        let firstWeight = weightHistory.length > 0 ? weightHistory[0] : null;

        if (!firstWeight) {
            firstWeight = {
                weight: Number(user.weight) || 0
            };
        }

        const initialWeight = Number(firstWeight.weight) || 0;
        // משקל נוכחי: הערך השמור אצל המשתמש, ואם אין - השקילה האחרונה בהיסטוריה
        const lastRecord = weightHistory.length > 0 ? weightHistory[weightHistory.length - 1] : null;
        const currentWeight = Number(user.weight) || Number(lastRecord?.weight) || 0;

        const weightChange = Number(
            (currentWeight - initialWeight).toFixed(1)
        );

        // חישוב הפרש מהשקילה הקודמת (דרישה 1)
        let previousWeightDiff = 0;
        if (weightHistory.length >= 2) {
            const prevW = Number(weightHistory[weightHistory.length - 2].weight);
            const currentW = Number(weightHistory[weightHistory.length - 1].weight);
            previousWeightDiff = Number((currentW - prevW).toFixed(1));
        }


        // ========================================
        // היסטוריית פעילויות - 4 בלבד
        // ========================================
        const activityHistory = await Activity.find({
            userId
        })
        .sort({
            date: -1
        })
        .limit(4);


        // ========================================
        // היסטוריית אימונים
        // ========================================
        const history = await Tracking.find({
            userId
        })
        .populate(
            'workoutId',
            'title date time instructor category estimatedCalories'
        )
        .sort({
            completedAt: -1
        })
        .limit(10);


        // ========================================
        // תשובה
        // ========================================
        res.status(200).json({
            weight: currentWeight,
            initialWeight,
            weightChange,
            previousWeightDiff, // הפרש מהשקילה הקודמת

            weeklyFoodCalories: foodCalories,

            weeklyWorkoutCalories: workoutCalories,
            weeklyWorkoutMinutes: workoutMinutes,
            weeklyWorkouts: weeklyWorkouts.length,

            weeklyActivityCalories: activityCalories,
            weeklyActivityMinutes: activityMinutes,
            weeklyActivities: weeklyActivities.length,

            totalBurnedCalories,
            calorieBalance,

            activityMinutes: workoutMinutes + activityMinutes,

            history,
            activityHistory
        });

    } catch (error) {
        next(error);
    }
};

import User from '../models/User.js';
import WeightHistory from '../models/WeightHistory.js';

// ========================================
// הוספת / עדכון משקל (המשקל נבדק ב-Joi)
// ========================================
export const addWeight = async (req, res, next) => {
    try {
        const userId = req.user.userId;
        const weight = req.body.weight;

        const user = await User.findById(userId);
        if (!user) {
            return res.status(404).json({
                message: 'המשתמש לא נמצא.'
            });
        }

        // 1. השקילה האחרונה והראשונה *לפני* ההוספה הנוכחית
        const lastRecord = await WeightHistory.findOne({ userId }).sort({ date: -1, _id: -1 });
        let firstWeight = await WeightHistory.findOne({ userId }).sort({ date: 1, _id: 1 });

        // 2. משתמש שכבר היה לו משקל שמור אבל אין היסטוריה - שומרים אותו כנקודת התחלה.
        //    (אם אין שום משקל קודם, השקילה הנוכחית היא נקודת ההתחלה ולא יוצרים רשומה כפולה)
        if (!firstWeight && user.weight) {
            firstWeight = await WeightHistory.create({
                userId,
                weight: Number(user.weight),
                date: user.createdAt || new Date()
            });
        }

        // השקילה הקודמת להשוואה (אם אין היסטוריה, נקודת ההתחלה שנוצרה הרגע)
        const previousRecord = lastRecord || firstWeight;

        // 3. עדכון המשתמש
        user.weight = weight;
        await user.save();

        // 4. שמירת ההיסטוריה החדשה
        const weightRecord = await WeightHistory.create({
            userId,
            weight,
            date: new Date()
        });

        if (!firstWeight) {
            firstWeight = weightRecord;
        }

        // 5. חישוב השינוי מהמשקל ההתחלתי ומהשקילה הקודמת
        const weightChange = Number((weight - Number(firstWeight.weight)).toFixed(1));

// אימות משתמש מחובר
export const verifyToken = (req, res, next) => {
    try {
        const authHeader = req.headers.authorization;

        if (!authHeader || !authHeader.startsWith('Bearer ')) {
            return res.status(401).json({
                message: 'גישה נדחתה. לא נמצא טוקן אימות.'
            });
        }

        const token = authHeader.split(' ')[1];

        if (!token) {
            return res.status(401).json({
                message: 'גישה נדחתה. לא נמצא טוקן אימות.'
            });
        }

        const secret = process.env.JWT_SECRET;

        if (!secret) {
            console.error('שגיאה: JWT_SECRET לא מוגדר בקובץ .env');

            return res.status(500).json({
                message: 'שגיאה פנימית בשרת.'
            });
        }

        const decoded = jwt.verify(token, secret);

        req.user = decoded;

        next();
    } catch (error) {
        return res.status(403).json({
            message: 'הטוקן אינו תקין או שפג תוקפו.'
        });
    }
};

import multer from 'multer';

const errorHandler = (err, req, res, next) => {
    // רישום פרטי השגיאה בטרמינל לצורכי איתור תקלות
    console.error('❌ Global Error Caught:', err.stack || err.message);

    // קביעת קוד סטטוס תקין
    let statusCode =
        Number.isInteger(err.statusCode) &&
        err.statusCode >= 400 &&
        err.statusCode <= 599
            ? err.statusCode
            : 500;

    let clientMessage = err.message;

    // שגיאות של העלאת קבצים (למשל קובץ גדול מדי)
    if (err instanceof multer.MulterError) {
        statusCode = 400;
        clientMessage =
            err.code === 'LIMIT_FILE_SIZE'
                ? 'הקובץ גדול מדי. הגודל המרבי הוא 5MB.'
                : 'שגיאה בהעלאת הקובץ.';
    }

    // שגיאת ולידציה של Mongoose (שדה חובה חסר, ערך מחוץ לטווח וכו')
    if (err.name === 'ValidationError' && err.errors) {
        statusCode = 400;
        clientMessage = Object.values(err.errors)
            .map(e => e.message)
            .join(', ');
    }

    // מזהה לא תקין (ObjectId)
    if (err.name === 'CastError') {
        statusCode = 400;
        clientMessage = 'מזהה לא תקין.';
    }

    // ערך כפול באינדקס ייחודי (למשל אימייל שכבר קיים)
    if (err.code === 11000) {
        statusCode = 400;
        clientMessage = 'הערך כבר קיים במערכת.';
    }

    // הודעה כללית כברירת מחדל
    let message = 'שגיאת שרת פנימית';

    // שגיאות צפויות שאפשר להחזיר ללקוח
    if (statusCode < 500 && clientMessage) {
        message = clientMessage;
    }

    // החזרת תשובת JSON
    res.status(statusCode).json({
        success: false,
        status: statusCode,
        message,
        ...(process.env.NODE_ENV === 'development' && {
            stack: err.stack
        })
    });
};

export default errorHandler;

const __dirname = path.dirname(fileURLToPath(import.meta.url));
export const UPLOADS_DIR = path.join(__dirname, '..', 'uploads');

// התיקייה לא נשמרת ב-Git (.gitignore), לכן יוצרים אותה אם היא חסרה - אחרת ב-Render ההעלאה תיכשל

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, UPLOADS_DIR);
  },

  filename: (req, file, cb) => {
    const uniqueSuffix =
      Date.now() + '-' + Math.round(Math.random() * 1E9);

    cb(
      null,
      uniqueSuffix + path.extname(file.originalname).toLowerCase()
    );
  }
});

const ALLOWED_MIME_TYPES = ['image/jpeg', 'image/png', 'image/webp'];

// מאפשרים רק תמונות
const fileFilter = (req, file, cb) => {
  if (ALLOWED_MIME_TYPES.includes(file.mimetype)) {
    return cb(null, true);
  }

  const error = new Error('ניתן להעלות רק תמונות מסוג JPG, PNG או WEBP.');
  error.statusCode = 400;
  cb(error);
};

const upload = multer({
  storage,
  fileFilter,
  limits: {
    fileSize: 5 * 1024 * 1024 // 5MB
  }
});

import fs from 'fs';

// Middleware אחד לכל הוולידציות עם Joi (מחליף 5 פונקציות כמעט זהות)
// source: 'body' (ברירת מחדל) או 'params'
export const validate = (schema, source = 'body') => {
    return (req, res, next) => {

        // אם לא נשלח גוף בכלל, בודקים אובייקט ריק כדי שהשדות החובה ייתפסו
        const { error, value } = schema.validate(req[source] ?? {}, {
            abortEarly: false,
            convert: true,       // המרות כמו "70" -> 70, וחיתוך רווחים
            stripUnknown: true   // שדות שלא מוגדרים בסכמה נזרקים (הגנה מפני שדות מיותרים)
        });

        if (error) {
            // אם הועלתה תמונה והבדיקה נכשלה - לא משאירים קובץ יתום בשרת
            if (req.file?.path) {
                fs.unlink(req.file.path, () => {});
            }

            return res.status(400).json({
                message: error.details.map(detail => detail.message).join(', ')
            });
        }

        req[source] = value;
        next();
    };
};

# VitaliFit API

VitaliFit is a robust Express backend API designed for comprehensive health and fitness management. It handles user accounts, workouts, nutrition analysis, activity logging, weight tracking, and reviews using a MongoDB database.

## ✨ Key Features

* **User Authentication:** Secure access control using JSON Web Tokens (JWT).
* **AI-Powered Nutrition:** Integrates the Gemini API to analyze meal images automatically.
* **Progress Tracking:** Dedicated modules for weight monitoring and physical activity logging.
* **System Monitoring:** Built-in `GET /api/status` endpoint to verify service health.

---

## 🚀 Getting Started

Follow these steps to set up and run the server in a local development environment.

### 1. Configure Environment Variables
Create a `.env` file in the `server` directory and define the following variables:

* **`MONGO_URI`** *(Required)*: The MongoDB connection string. The server will not start without this.
* **`JWT_SECRET`** *(Required)*: A strong, private secret key used to sign and verify authentication tokens.
* **`GEMINI_API_KEY`** *(Required)*: The API key needed to process AI-powered meal image analysis.
* **`PORT`** *(Optional)*: The network port the server will listen on (Defaults to `5000`).
* **`NODE_ENV`** *(Optional)*: The runtime environment. Set to `development` to expose error stack traces.

### 2. Install and Run
Open your terminal, navigate to the server directory, install the required packages, and launch the application:

const BASE = process.env.BASE_URL || 'http://localhost:5000';
const stamp = Date.now();
const email = `smoke-${stamp}@example.com`;
const password = 'Test1234';

let passed = 0;
let failed = 0;

async function call(method, path, { token, body } = {}) {
    const res = await fetch(`${BASE}${path}`, {
        method,
        headers: {
            ...(body ? { 'Content-Type': 'application/json' } : {}),
            ...(token ? { Authorization: `Bearer ${token}` } : {})
        },
        body: body ? JSON.stringify(body) : undefined
    });

    let data = null;
    try { data = await res.json(); } catch { /* לא JSON */ }
    return { status: res.status, data };
}

function check(name, condition, details = '') {
    if (condition) {
        passed++;
        console.log(`  ✅ ${name}`);
    } else {
        failed++;
        console.log(`  ❌ ${name} ${details}`);
    }
}

const section = (title) => console.log(`\n▶ ${title}`);

async function run() {
    section('שרת');
    const status = await call('GET', '/api/status');
    if (status.status === 403) {
        console.log('  ⚠️  השרת חסום היום (חסימת שבת). הריצי שוב ביום אחר.');
        process.exit(2);
    }
    check('GET /api/status מחזיר 200', status.status === 200, `(קיבלתי ${status.status})`);

    const unknown = await call('GET', '/api/does-not-exist');
    check('נתיב לא קיים מחזיר 404 בפורמט JSON', unknown.status === 404 && !!unknown.data?.message);

    section('הרשמה והתחברות');
    const badReg = await call('POST', '/api/auth/register', { body: { fullName: 'א', email: 'not-an-email', password: '1' } });
    check('הרשמה עם נתונים לא תקינים נדחית (400)', badReg.status === 400, `(קיבלתי ${badReg.status})`);

    const emptyReg = await call('POST', '/api/auth/register', {});
    check('הרשמה בלי גוף בכלל נדחית (400)', emptyReg.status === 400, `(קיבלתי ${emptyReg.status})`);

    const reg = await call('POST', '/api/auth/register', { body: { fullName: 'בדיקה אוטומטית', email, password } });
    check('הרשמה תקינה (201) ומחזירה טוקן', reg.status === 201 && !!reg.data?.token, `(קיבלתי ${reg.status})`);
    check('התשובה לא כוללת סיסמה', !JSON.stringify(reg.data || {}).includes('password'));

    const dup = await call('POST', '/api/auth/register', { body: { fullName: 'בדיקה אוטומטית', email, password } });
    check('הרשמה כפולה נדחית (400)', dup.status === 400);

    const wrongPass = await call('POST', '/api/auth/login', { body: { email, password: 'wrong-password' } });
    check('התחברות עם סיסמה שגויה נדחית (400)', wrongPass.status === 400);

    const login = await call('POST', '/api/auth/login', { body: { email, password } });
    check('התחברות תקינה (200)', login.status === 200 && !!login.data?.token);
    const token = login.data?.token;

    const noToken = await call('GET', '/api/auth/profile');

import { GoogleGenAI } from '@google/genai';
import fs from 'fs/promises';

const ai = new GoogleGenAI({
    apiKey: process.env.GEMINI_API_KEY
});


async function analyzeMealImage(imagePath, mimeType, mealName) {

    const imageData = await fs.readFile(imagePath);
    const base64Image = imageData.toString('base64');


    const interaction = await ai.interactions.create({

        model: 'gemini-3.5-flash-lite',

        input: [

            {
                type: 'image',
                mime_type: mimeType,
                data: base64Image
            },

            {
                type: 'text',
                text: `
                    נתח את הארוחה שבתמונה.

                    שם הארוחה שהמשתמש הזין:
                    ${mealName}

                    הערך את הכמות הנראית בתמונה
                    והחזר הערכה תזונתית עבור כל הארוחה.

                    חשוב:
                    הערכים הם הערכה בלבד לפי התמונה,
                    ולא מדידה מדויקת.
                `
            }

        ],

        response_format: {

            type: 'text',
            mime_type: 'application/json',

            schema: {

                type: 'object',

                properties: {

                    mealName: {
                        type: 'string'
                    },

                    calories: {
                        type: 'number'
                    },

                    protein: {
                        type: 'number'
                    },

                    carbs: {
                        type: 'number'
                    },

                    fats: {
                        type: 'number'
                    }

                },

                required: [
                    'mealName',
                    'calories',
                    'protein',
                    'carbs',
                    'fats'
                ]

            }

        },

        store: false
    });


    return JSON.parse(interaction.output_text);
}


export default analyzeMealImage;

export function calculateCalories(weight, durationMinutes, met) {
  return Math.round((met * 3.5 * weight / 200) * durationMinutes);
}

// תחילת השבוע הנוכחי (יום ראשון, 00:00)
export function getStartOfWeek(date) {
    const result = new Date(date);
    result.setHours(0, 0, 0, 0);
    result.setDate(result.getDate() - result.getDay());
    return result;
}

import jwt from 'jsonwebtoken';

// יצירת טוקן התחברות - במקום קוד כפול בהרשמה ובהתחברות
export function signToken(user) {
    if (!process.env.JWT_SECRET) {
        throw new Error('JWT_SECRET לא מוגדר בקובץ .env');
    }

    return jwt.sign(
        { userId: user._id, role: user.role },
        process.env.JWT_SECRET,
        { expiresIn: '1d' }
    );
}

// הפרטים של המשתמש שמותר להחזיר ללקוח (בלי סיסמה)
export function publicUser(user) {
    return {
        id: user._id,
        fullName: user.fullName,
        email: user.email,
        role: user.role
    };
}
